# Guardian Network Production Deployment Guide

## Prerequisites

### Required Infrastructure

- ✅ **PostgreSQL 12+** with connection pooling
- ✅ **Redis 6+** for caching (optional but recommended)
- ✅ **Node.js 18+** with TypeScript support
- ✅ **HTTPS endpoints** for secure communication
- ✅ **Monitoring stack** (Prometheus + Grafana recommended)

### Required Environment Variables

```bash
# Guardian Network Core
GUARDIAN_NETWORK_API_KEY=<your-api-key-from-omsystems>
GUARDIAN_NETWORK_SALT=<unique-salt-for-your-deployment>
```

## Step 2: Application Integration

### Add to Your Main Application (app.ts)

```typescript
import type { Pool } from 'pg';
import type { Redis } from 'ioredis';
import { 
  initializeGuardianNetwork, 
  registerGuardianNetworkRoutes,
  shutdownGuardianNetwork 
} from './intelligence';

// In your application startup
async function startApplication() {
  // ... your existing startup code ...
  
  // Get your existing database pool
  const pool: Pool = (store as any).pool || (store as any).db;
  
  // Get Redis if available
  const redis: Redis | undefined = yourRedisInstance;  // optional
  
  // Initialize Guardian Network
  console.log('Initializing Guardian Network...');
  const guardianNetwork = await initializeGuardianNetwork(pool, redis);
  
  // Register routes
  await registerGuardianNetworkRoutes(fastify, guardianNetwork);
  
  // Setup event handlers
  guardianNetwork.on('threat-alert', (alert) => {
    console.log(`🚨 THREAT ALERT: ${alert.title}`);
    // Send to SOC, mobile push, SMS, etc.
    notifySecurityTeam(alert);
  });
  
  guardianNetwork.on('pattern-matched', (match) => {
    console.log(`✅ Pattern matched: ${match.intelligence.previousOccurrences} occurrences`);
    // Attach intelligence to incident UI
    enrichIncidentWithIntelligence(match.localIncidentId, match);
  });
  
  // Graceful shutdown
  process.on('SIGTERM', async () => {
    await shutdownGuardianNetwork();
    process.exit(0);
  });
}
```

### Auto-Share Verified Incidents

Integrate with your incident resolution workflow:

```typescript
import { getGuardianNetworkService } from './intelligence';

// When an incident is verified (true positive)
async function resolveIncident(incidentId: string, resolution: 'true-positive' | 'false-positive') {
  // ... your existing resolution logic ...
  
  if (resolution === 'true-positive') {
    // Get incident details
    const incident = await getIncidentFromDatabase(incidentId);
    
    // Convert to LocalIncident format
    const localIncident = {
      id: incident.id,
      tenantId: incident.tenantId,
      branchId: incident.branchId,
      detectionType: incident.detectionType,
      severity: incident.severity,
      category: incident.category,
      occurredAt: incident.occurredAt,
      detectedAt: incident.detectedAt,
      cameraId: incident.cameraId,
      cameraName: incident.cameraName,
      aiCapabilities: incident.aiCapabilities || [],
      confidence: incident.confidence || 0.85,
      responseTime: incident.responseTime,
      outcome: incident.outcome || 'detected-after',
      industryVertical: 'banking',
      facilityType: 'branch',
    };
    
    // Auto-share with network
    try {
      const guardianNetwork = getGuardianNetworkService();
      const result = await guardianNetwork.shareIncident(localIncident);
      
      if (result.success) {
        console.log(`✅ Incident ${incidentId} shared as pattern ${result.patternId}`);
      }
    } catch (error) {
      console.error('Failed to share incident:', error);
    }
  }
}
```

### Auto-Match New Incidents

Integrate with your alert creation workflow:

```typescript
import { getGuardianNetworkService } from './intelligence';

// When a new alert is created
async function createAlert(alertData: any) {
  // ... create alert in database ...
  
  // Auto-match against global patterns
  try {
    const guardianNetwork = getGuardianNetworkService();
    
    const localIncident = {
      id: alert.id,
      tenantId: alert.tenantId,
      branchId: alert.branchId,
      detectionType: alert.detectionType,
      severity: alert.severity,
      category: alert.category,
      occurredAt: alert.occurredAt,
      detectedAt: alert.detectedAt,
      cameraId: alert.cameraId,
      cameraName: alert.cameraName,
      aiCapabilities: alert.aiCapabilities || [],
      confidence: alert.confidence || 0.85,
      outcome: 'unknown',  // Not resolved yet
      industryVertical: 'banking',
      facilityType: 'branch',
    };
    
    const match = await guardianNetwork.matchIncidentToPatterns(localIncident);
    
    if (match) {
      console.log(`🎯 Pattern match! Similarity: ${match.matchedPatterns[0].similarityScore}`);
      console.log(`📊 Intelligence:`, match.intelligence);
      console.log(`💡 Recommendations:`, match.recommendedActions);
      
      // Store intelligence in database
      await storeIntelligenceForIncident(alert.id, match);
      
      // Display in UI
      await notifyOperatorWithIntelligence(alert.id, match);
      
      // Auto-apply immediate actions
      for (const action of match.recommendedActions.filter(a => a.priority === 'immediate')) {
        await executeRecommendedAction(alert.id, action);
      }
    }
  } catch (error) {
    console.error('Pattern matching failed:', error);
  }
  
  return alert;
}
```

## Step 5: Security Hardening

### 1. Network Security

```bash
# Firewall rules (iptables example)
# Allow outbound HTTPS to Guardian Network hub
iptables -A OUTPUT -p tcp --dport 443 -d guardian-network.omsystems.ai -j ACCEPT

# Allow outbound WebSocket
iptables -A OUTPUT -p tcp --dport 443 -d guardian-network.omsystems.ai -j ACCEPT

# Block all other outbound to unknown IPs
```

### 2. API Key Rotation

```bash
# Generate new API key monthly
npm run guardian-network:rotate-key

# Update environment variable
export GUARDIAN_NETWORK_API_KEY=<new-key>

# Restart application
pm2 restart omsystems
```

### 3. Database Security

```sql
-- Create dedicated role
CREATE ROLE guardian_network_app WITH LOGIN PASSWORD '<strong-password>';

-- Grant minimal permissions
GRANT USAGE ON SCHEMA guardian_network TO guardian_network_app;
GRANT SELECT, INSERT, UPDATE ON ALL TABLES IN SCHEMA guardian_network TO guardian_network_app;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA guardian_network TO guardian_network_app;

-- Revoke dangerous permissions
REVOKE DELETE, TRUNCATE, DROP ON ALL TABLES IN SCHEMA guardian_network FROM guardian_network_app;
```
